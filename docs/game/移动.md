```
// 技能会根据当前方向确定伤害区域
/
     * 计算角度 (0-360度)
     * 从当前位置指向目标位置的角度, 上方为0度 顺时针
     * @param currentPos 当前位置
     * @param targetPos 目标位置
     * @returns 角度值 (0-360度)
     */
    static calculateAngleCW(currentPos: cc.Vec3, targetPos: cc.Vec3): number {
        // 计算方向向量
        let dx = targetPos.x - currentPos.x;
        let dy = targetPos.y - currentPos.y;

        return MathUtils.vectorToAngleCW(dx, dy);
    }


    / 向量转顺时针角度 上方为0度 顺时针旋转 */
    static vectorToAngleCW(x: number, y: number): number {
        // 使用Math.atan2(x, y)可以得到上方(Y+)为0度，顺时针(CW)旋转的结果
        let radians = Math.atan2(x, y);
        let degrees = radians * (180 / Math.PI);
        if (degrees < 0) {
            degrees += 360;
        }
        return degrees;
    }
```
